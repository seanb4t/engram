   ```go
   type VersionBucket struct {
       Version int    `json:"version"`
       Count   uint64 `json:"count"`
   }
   type MigrateStatusResult struct {
       Buckets     []VersionBucket `json:"buckets"`      // 0 <= Version <= CurrentVersion, ascending
       Absent      uint64          `json:"absent"`       // no schema_version key at all (legacy)

   ```
